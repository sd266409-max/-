const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const cors = require('cors'); const path = require('path'); const app = express(); const server = http.createServer(app); const PORT = process.env.PORT || 3000; app.use(cors()); app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); // 현재 접속 중인 플레이어 목록 const activePlayers = {}; io = new Server(server, { cors: { origin: "*", methods: ["GET", "POST"] } }); io.on('connection', (socket) => { console.log(`[접속] 플레이어 입장: ${socket.id}`); // 게임 바로 입장 socket.on('joinGame', (data) => { activePlayers[socket.id] = { id: socket.id, nickname: data?.nickname || `낚시꾼_${socket.id.substring(0, 4)}`, gold: 50, position: data?.position || { x: 0, y: 0.8, z: 4 }, rotationY: 0 }; // 다른 유저에게 알림 및 전체 유저 데이터 전송 socket.broadcast.emit('playerJoined', activePlayers[socket.id]); socket.emit('currentPlayers', activePlayers); }); // 위치 이동 동기화 socket.on('playerMove', (movementData) => { if (activePlayers[socket.id]) { activePlayers[socket.id].position = movementData.position; activePlayers[socket.id].rotationY = movementData.rotationY; socket.broadcast.emit('playerMoved', { id: socket.id, position: movementData.position, rotationY: movementData.rotationY }); } }); // 퇴장 처리 socket.on('disconnect', () => { console.log(`[퇴장] 플레이어 퇴장: ${socket.id}`); delete activePlayers[socket.id]; io.emit('playerLeft', socket.id); }); }); app.get('*', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); server.listen(PORT, () => { console.log(`3D 낚시 게임 서버가 포트 ${PORT}에서 실행 중입니다.`); });
